\begin{frame}{En pratique : estimer la densité}
    \begin{center}
        \formulebox{\Large\mathversion{bold} $\widehat{f} \leftarrow \widehat{f}_{K\text{-NN}}$} \\[4pt]
        \normalsize \textbf{Estimateur des $K$ plus proches voisins} \citb{biau15}
    \end{center}

    \vspace{0.4cm}

    \begin{columns}[c, onlytextwidth]
        \begin{column}{0.56\textwidth}
            \begin{center}
                {\large\mathversion{bold} $\forall y \in \mathbb{R}^p, \quad \widehat{f}_{K\text{-NN}}(y) = \dfrac{K}{n \, \omega_p \, r_K(y)^p}$}

                \vspace{1.0cm}

                \formulebox{\small\mathversion{bold} $r_K(y) = \min \left\{ r \ge 0 \;\middle|\; |B(y,r) \cap (\mathcal{X}\setminus\{y\})| \ge K \right\}$}
            \end{center}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{itemize}
                \setlength{\itemsep}{8pt}
                \item \textbf{$y$} : un point de l'\textbf{espace}, du nuage ou non
                \item \textbf{$n$} : nombre de points de $\mathcal{X}$
                \item \textbf{$\omega_p$} : volume de la boule unité de $\mathbb{R}^p$
                \item \textbf{$r_K(y)$} : distance de $y$ à son $K$\up{e} plus proche voisin ; si $y\in\mathcal X$, on exclut $y$ lui-même
            \end{itemize}
        \end{column}

\begin{frame}[t]{Comment guider l'attention \emph{via} la hiérarchie ? Une idée à explorer}
    \vspace{0.02cm}
    \begin{center}
        \resizebox{0.64\textwidth}{!}{\input{figs/attention_biais}}
    \end{center}

    \vspace{0.06cm}
    \small
    \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item L'\textbf{attention} : chaque jeton pose une \textbf{requête}, la compare aux
        \textbf{clefs} de tous les autres, et repart avec la moyenne de leurs \textbf{valeurs},
        pondérée par un \textbf{score visuel} que SAM sait déjà calculer
        \item On y \textbf{ajoute un bonus} avant le \emph{softmax} : un échange \textbf{favorisé},
        jamais imposé \citb{ying21}
        \item Le bonus vient de la \textbf{hiérarchie} du graphe de \textsc{Frangi} : plus deux jetons
        \textbf{fusionnent tôt} dans l'arbre, plus il est fort, même loin dans l'image \citb{euvip26} ;
        LoRA l'apprend avec un \textbf{seul} coefficient $\beta$ \citb{hu22}
    \end{itemize}

    \vspace{0.06cm}
    \begin{center}
        \formulebox{\parbox{0.80\textwidth}{\centering\small
        $\mathrm{attention} \;=\; \mathrm{softmax}\bigl(\text{scores visuels} \;+\; \beta\, B_H\bigr)$}}
    \end{center}
    \reffoot{ying21,euvip26,hu22}
\end{frame}

\section{Conclusion}
\frame{\sectionpage}

\begin{frame}[t]{Deux fils conducteurs, quatre contributions}
    \vspace{-0.10cm}
    \begin{beamercolorbox}[rounded=true,sep=3pt]{block body}
        \centering\small
        \textbf{1)} Quand les relations binaires sont insuffisantes ou instables,\\ les \textbf{interactions d'ordre supérieur} apportent une cohérence robuste.\\[3pt]
        \textbf{2)} La \textbf{percolation} est la clef de voûte théorique\\ pour quantifier les performances de ces algorithmes
    \end{beamercolorbox}

    \vspace{0.06cm}
    \footnotesize
    \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item Une \textbf{vue unifiée} du \emph{Single-Linkage} : heuristique, géométrique, statistique \citb{ans26} ; LiDAR 3D et 4D sans apprentissage
        \item \textbf{HGP-Clusterer} : la hiérarchie $K$-NN \emph{exacte}, \emph{via} \v{C}ech, \textsc{Gabriel} et \textsc{Delaunay} d'ordre $K$ \citb{eusipco24} \citb{cn24} \citb{ans26}
        \item La \textbf{vitesse de percolation} : elle explique l'échec de Robust SL et (H)DBSCAN, et les gains des $K$-polyèdres \citb{src24}
        \item L'\textbf{export du paradigme} : graphes signés \citb{asilomar25} \citb{ii26} ; imagerie, graphe de \textsc{Frangi} multimodal \citb{gretsi25} \citb{euvip26} \citb{isprs26}
    \end{itemize}
    \reffoot{eusipco24,cn24,ans26,src24,asilomar25,ii26,gretsi25,euvip26,isprs26}
\end{frame}

\begin{frame}{Autres perspectives}
    \begin{itemize}
        \setlength{\itemsep}{11pt}
        \item \textbf{La grande dimension} : une \textbf{réduction de dimension} d'ordre supérieur,
        là où les méthodes courantes s'appuient sur des graphes de voisinage (UMAP)
        \item \textbf{Question ouverte} : HGP-Clusterer n'est que \emph{fractionnellement} consistant ;
        un algorithme consistant en dimension $p \ge 2$ reste à inventer
    \end{itemize}
\end{frame}
