\begin{frame}{In practice: estimating the density}
    \begin{center}
        \formulebox{\Large\mathversion{bold} $\widehat{f} \leftarrow \widehat{f}_{K\text{-NN}}$} \\[4pt]
        \normalsize \textbf{$K$-nearest neighbour estimator} \citb{biau15}
    \end{center}

    \vspace{0.4cm}

    \begin{columns}[c, onlytextwidth]
        \begin{column}{0.56\textwidth}
            \begin{center}
                {\large\mathversion{bold} $\forall y \in \mathbb{R}^p, \quad \widehat{f}_{K\text{-NN}}(y) = \dfrac{K}{n \, \omega_p \, r_K(y)^p}$}

                \vspace{1.0cm}

                \formulebox{\small\mathversion{bold} $r_K(y) = \min \left\{ r \ge 0 \;\middle|\; |B(y,r) \cap (\mathcal{X}\setminus\{y\})| \ge K \right\}$}
            \end{center}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{itemize}
                \setlength{\itemsep}{8pt}
                \item \textbf{$y$}: a point of the \textbf{space}, in the point cloud or not
                \item \textbf{$n$}: number of points in $\mathcal{X}$
                \item \textbf{$\omega_p$}: volume of the unit ball of $\mathbb{R}^p$
                \item \textbf{$r_K(y)$}: distance from $y$ to its $K$th nearest neighbour; if $y\in\mathcal X$, $y$ itself is excluded
            \end{itemize}
        \end{column}
    \end{columns}

\section{Conclusion}
\frame{\sectionpage}

\begin{frame}[t]{Two guiding threads, four contributions}
    \vspace{-0.10cm}
    \begin{beamercolorbox}[rounded=true,sep=3pt]{block body}
        \centering\small
        \textbf{1)} When pairwise relations are insufficient or unstable,\\ \textbf{higher-order interactions} bring robust coherence.\\[3pt]
        \textbf{2)} \textbf{Percolation} is the theoretical keystone\\ for quantifying the performance of these algorithms
    \end{beamercolorbox}

    \vspace{0.06cm}
    \footnotesize
    \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item A \textbf{unified view} of \emph{Single-Linkage}: heuristic, geometric, statistical \citb{ans26}; training-free 3D and 4D LiDAR
        \item \textbf{HGP-Clusterer}: the \emph{exact} $K$-NN hierarchy, \emph{via} \v{C}ech, \textsc{Gabriel} and order-$K$ \textsc{Delaunay} \citb{eusipco24} \citb{cn24} \citb{ans26}
        \item The \textbf{percolation rate}: it explains the failure of Robust SL and (H)DBSCAN, and the gains of $K$-polyhedra \citb{src24}
        \item \textbf{Exporting the paradigm}: signed graphs \citb{asilomar25} \citb{ii26}; imaging, multimodal \textsc{Frangi} graph \citb{gretsi25} \citb{euvip26} \citb{isprs26}
    \end{itemize}
    \reffoot{eusipco24,cn24,ans26,src24,asilomar25,ii26,gretsi25,euvip26,isprs26}
\end{frame}

\begin{frame}{Further perspectives}
    \begin{itemize}
        \setlength{\itemsep}{11pt}
        \item \textbf{High dimension}: a higher-order \textbf{dimensionality reduction},
        where current methods rely on neighbourhood graphs (UMAP)
        \item \textbf{Open question}: HGP-Clusterer is only \emph{fractionally} consistent;
        a consistent algorithm in dimension $p \ge 2$ remains to be invented
    \end{itemize}
\end{frame}
