\begin{frame}{En pratique : estimer la densité}
    \begin{center}
        \Large $\widehat{f} \leftarrow \widehat{f}_{K\text{-NN}}$ \\[4pt]
        \normalsize \textbf{Estimateur des $K$ plus proches voisins}
    \end{center}

    \vspace{0.4cm}

    \begin{columns}[c, onlytextwidth]
        \begin{column}{0.56\textwidth}
            \begin{center}
                \formulebox{\Large\mathversion{bold} $\forall x \in \mathbb{R}^p, \quad \widehat{f}_{K\text{-NN}}(x) = \dfrac{K}{n \, \omega_p \, r_K(x)^p}$}

                \vspace{1.0cm}

                \formulebox{\large\mathversion{bold} $r_K(x) = \min \left\{ r \ge 0 \;\middle|\; |B(x,r) \cap \mathcal{X}| \ge K \right\}$}
            \end{center}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{itemize}
                \setlength{\itemsep}{8pt}
                \item \textbf{$n$} : nombre de points de $\mathcal{X}$
                \item \textbf{$\omega_p$} : volume de la boule unité de $\mathbb{R}^p$
                \item \textbf{$r_K(x)$} : distance de $x$ à son $K$\up{e} plus proche voisin
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

        \begin{column}{0.48\textwidth}
            \textbf{Effet de chaînage} : une chaîne de points suffit à fusionner deux amas

            \vspace{0.4cm}
            \textbf{Robust Single-Linkage} \citb{chaudhuri10} :
            \begin{itemize}
                \setlength{\itemsep}{6pt}

\section{Conclusion \& perspectives}
\frame{\sectionpage}

\begin{frame}{Deux piliers, quatre contributions}
    \begin{beamercolorbox}[rounded=true,sep=8pt]{block body}
        \centering
        \textbf{1)} Quand les relations binaires sont insuffisantes ou instables,\\ les \textbf{interactions d'ordre supérieur} apportent une cohérence robuste.\\[6pt]
        \textbf{2)} La \textbf{percolation} est la clef de voûte théorique\\ pour quantifier les performances de ces algorithmes
    \end{beamercolorbox}

    \vspace{0.5cm}
    \begin{itemize}
        \setlength{\itemsep}{8pt}
        \item Une \textbf{vue unifiée} du Single-Linkage : heuristique (arbre couvrant), géométrique (percolation), statistique (\textsc{Hartigan}),\\ avec applications LiDAR 3D/4D sans apprentissage \citb{ans26}
        \item \textbf{HGP-Clusterer} : la hiérarchie $K$-NN \emph{exacte}, \emph{via} \v{C}ech, \textsc{Gabriel} et la mosaïque de \textsc{Delaunay} d'ordre $K$ \citb{eusipco24} \citb{cn24}
        \item La \textbf{vitesse de percolation} : une mesure quantitative qui explique l'échec de Robust SL / (H)DBSCAN et la supériorité des $K$-polyèdres \citb{src24}
        \item L'\textbf{export du paradigme} : graphes signés (dynamiques d'ordre supérieur, bornes plus fines) \citb{asilomar25} et imagerie (graphe de Frangi multimodal) \citb{gretsi25} \citb{euvip26}
    \end{itemize}
\end{frame}

\begin{frame}{Perspectives}
    \begin{itemize}
        \setlength{\itemsep}{12pt}
        \item \textbf{Réduction de dimension d'ordre supérieur} : la mosaïque de \textsc{Delaunay} d'ordre $K$ est bien plus riche que les graphes de voisinage (UMAP) : la piste privilégiée
        \item \textbf{Question ouverte assumée} : HGP-Clusterer n'est que \emph{fractionnellement} consistant ; un algorithme consistant en dimension $p \ge 2$ reste à inventer
        \item \textbf{Graphes signés} : couplage à deux répliques pour des seuils exacts de recouvrement ; vers un algorithme pratique de détection de communautés
        \item \textbf{Fissures} : un logiciel d'aide à l'annotation experte fondé sur les graphes ; réconcilier modèles géométriques et apprentissage profond
    \end{itemize}
\end{frame}

\begin{frame}{Le cadre bayésien et l'hypothèse de Gibbs}
    \begin{center}
        \begin{tikzpicture}[node distance=0.7cm and 0.9cm, scale=0.85, every node/.style={scale=0.85}]
            \node[chapbox] (prior) {A priori\\$\mu_{0}$};
            \node[chapbox,right=of prior] (truth) {Vérité cachée\\$\Sigma$};
            \node[chapbox,right=of truth] (chan) {Canal bruité\\$Q$};
            \node[chapboxdark,right=of chan] (obs) {Observation\\$(X,W)$};
            \node[chapbox,right=of obs] (post) {A posteriori\\$\mu_{X,W}$};
            \draw[chaparr] (prior) -- (truth);
            \draw[chaparr] (truth) -- (chan);
            \draw[chaparr] (chan) -- (obs);
            \draw[chaparr] (obs) -- (post);
        \end{tikzpicture}
    \end{center}

    \vspace{0.3cm}
    \begin{center}
        \formulebox{$U(\sigma) = \displaystyle\sum_{e \in F} W_e \ind{\sigma_i \neq \sigma_j} + \sum_{e \in \tilde F} |W_e| \ind{\sigma_i = \sigma_j}
        \qquad
        \mu_{X,W}(\sigma) = \frac{e^{-U(\sigma)}}{Z}$}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \item $F$ : arêtes attractives ($W_e > 0$) ; $\tilde F$ : répulsives ($W_e < 0$) ; $U$ pénalise les arêtes insatisfaites
        \item \textbf{Hypothèse de Gibbs} : la loi \emph{a posteriori} est la mesure de Gibbs signée
    \end{itemize}
\end{frame}

\begin{frame}{Un pas de Swendsen--Wang signé, en détail}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Gel} : chaque arête \emph{satisfaite} est gelée avec probabilité $1 - e^{-|W_e|}$ ; les insatisfaites jamais
        \item \textbf{Recoloriage} : chaque amas gelé est recolorié uniformément au hasard
        \item La dynamique laisse la loi \emph{a posteriori} invariante
        \item Un pas depuis la vérité $\Sigma$ produit une \textbf{réplique} $\Sigma^{(1)} \sim \mu_{X,W}$ (identité de Nishimori) : un couplage entre vérité et observation
        \item Les petits amas sont recoloriés aveuglément : l'information globale ne survit que dans les \textbf{composantes géantes}
    \end{itemize}
\end{frame}

\begin{frame}{Le modèle de Sankararaman--Baccelli, en détail}
    \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Nœuds : processus de Poisson dans $\mathbb R^p$ ; communautés $\Sigma_i \in \{\pm 1\}$ uniformes
        \item Arête $e = \{i,j\}$ présente avec probabilité $f_{\mathrm{in}}(r_e)$ (même communauté) ou $f_{\mathrm{out}}(r_e)$ (communautés différentes), $r_e = \|X_i - X_j\|$
        \item Poids signés reconstruits : $W_e = \log\frac{f_{\mathrm{in}}(r_e)}{f_{\mathrm{out}}(r_e)} \ge 0$ si $e$ observée, $W_e = -\log\frac{1-f_{\mathrm{out}}(r_e)}{1-f_{\mathrm{in}}(r_e)} \le 0$ sinon
    \end{itemize}

    \vspace{0.3cm}
    \begin{center}
        \formulebox{$\mathbb P(\text{arête gelée}) = f_{\mathrm{in}}(r_e) \left(1 - e^{-|W_e|}\right) = f_{\mathrm{in}}(r_e) - f_{\mathrm{out}}(r_e)$}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \item Le graphe gelé d'un pas de SW-arêtes est \textbf{exactement} la percolation indépendante de \citb{sb18}
    \end{itemize}
\end{frame}

\begin{frame}{La hiérarchie d'horloges, en détail}
    \begin{itemize}
        \setlength{\itemsep}{8pt}
        \item Conditionnellement à $\sigma$ : $\xi_e \sim \mathrm{Exp}(|W_e|)$ sur les arêtes satisfaites, $\xi_e = +\infty$ sinon ;\\
        $\Pi_\beta$ = composantes de $\{\xi_e \le \beta\}$ ; le dendrogramme s'obtient par \textsc{Kruskal}
        \item $\mathbb P(\xi_e \le 1) = 1 - e^{-|W_e|}$ : $\Pi_1$ est \textbf{exactement} la partition gelée de Swendsen--Wang
        \item Mesure jointe de type \textsc{Edwards}--\textsc{Sokal} $\nu(\sigma, D)$ ; à chaque nœud, un \textbf{heat bath exact} sur les quatre orientations relatives des deux fils (balance détaillée, invariance)
        \item Le dendrogramme est \textbf{non marqué} : l'identité de l'arête gagnante est marginalisée --- une fusion dépend de toute la coupe $\Lambda_u(\sigma) = \sum_{e \in E_u} |W_e| \ind{e \text{ satisfaite}}$
        \item Sur le tore triangulaire : $\Pi_t$ suit une percolation indépendante $q_p(t) = p\,(1 - e^{-u_p t})$, $u_p = \log\frac{p}{1-p}$
    \end{itemize}
\end{frame}
