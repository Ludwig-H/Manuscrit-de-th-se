% 7.a — Retour au point de départ : Kleinberg, puis Carlsson--Mémoli
% ============================================================
\begin{frame}{Retour au début : le théorème d'impossibilité de \textsc{Kleinberg}}
    \begin{columns}[c, onlytextwidth]
        \begin{column}{0.60\textwidth}
            \small
            Une fonction de clustering $\mathcal C$ associe à un espace métrique fini
            $(\mathcal X, d)$, avec $|\mathcal X| \ge 2$, une \textbf{partition} de $\mathcal X$.

            \vspace{0.25cm}
            \begin{enumerate}[(i)]
                \setlength{\itemsep}{7pt}
                \item \textbf{Invariance d'échelle} : $\mathcal C(\mathcal X, \lambda d) = \mathcal C(\mathcal X, d)$
                pour tout $\lambda > 0$ --- changer d'unité ne change rien.
                \item \textbf{Richesse} : toute partition de $\mathcal X$ est atteinte par une distance bien choisie.
                \item \textbf{Cohérence} : resserrer un amas \emph{rendu par $\mathcal C$}, ou écarter deux
                de ces amas, laisse le résultat inchangé.
            \end{enumerate}
        \end{column}
        \begin{column}{0.38\textwidth}
            \centering
            \resizebox{0.86\textwidth}{!}{\input{figs/kleinberg_axiomes}}
        \end{column}
    \end{columns}

    \vspace{0.30cm}
    \begin{center}
        \formulebox{\textbf{Impossibilité} \citb{kleinberg02} : \emph{aucune} fonction de clustering ne satisfait ces trois axiomes.}
    \end{center}
    \reffoot{kleinberg02}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{\textsc{Carlsson} \& \textsc{Mémoli} : changer l'objet de sortie}
    \begin{center}
        \resizebox{0.57\textwidth}{!}{\input{figs/kleinberg_coupes}}
    \end{center}

    \vspace{0.05cm}
    \small
    \begin{itemize}
        \setlength{\itemsep}{5pt}
        \item Pour chacune des trois \emph{paires} d'axiomes, \textsc{Kleinberg} donne un témoin ---
        à chaque fois une \textbf{variante du Single-Linkage}, la variante étant la \textbf{règle d'arrêt} :
        $k$ amas, distance $r$, ou fraction $\alpha$ du diamètre
        \item L'obstruction indique donc la sortie : garder l'arbre, et laisser la règle de coupe à l'utilisateur
        \item \textsc{Carlsson} \& \textsc{Mémoli} changent alors l'objet rendu : une \textbf{ultramétrique},
        $T : (\mathcal X, d) \mapsto (\mathcal X, u)$ --- de façon équivalente, un dendrogramme
    \end{itemize}

    \vspace{0.10cm}
    \begin{center}
        \formulebox{\parbox{0.82\textwidth}{\centering\small
        Sous trois axiomes adaptés (normalisation \emph{à deux points}, fonctorialité, séparation),\\
        \textbf{le Single-Linkage est la seule solution} \citb{carlsson10}.}}
    \end{center}
    \reffoot{carlsson10}
\end{frame}

% ============================================================
% 7.b — La dynamique hiérarchique : une horloge par arête
% ============================================================
\begin{frame}{Après \textsc{Swendsen}--\textsc{Wang} : une \emph{horloge} par arête}
    \small
    Un pas de \textsc{Swendsen}--\textsc{Wang} gèle chaque arête satisfaite avec probabilité
    $1 - e^{-|W_e|}$, puis recolorie les amas gelés. Remplaçons ce tirage \emph{binaire} par un \textbf{temps}.

    \vspace{0.25cm}
    \begin{center}
        \formulebox{\parbox{0.86\textwidth}{\centering
        $\xi_e \sim \mathrm{Exp}\bigl(|W_e|\bigr)$ si $e$ est satisfaite, \quad $\xi_e = +\infty$ sinon ;\\[5pt]
        $\Pi_\beta \;=\;$ composantes connexes du graphe $\bigl(V, \{e : \xi_e \le \beta\}\bigr)$,
        \qquad $0 \le \beta \le 1$.}}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Comme $\mathbb P(\xi_e \le 1 \mid e \text{ satisfaite}) = 1 - e^{-|W_e|}$, la coupe à $\beta = 1$
        redonne \textbf{exactement} la partition gelée de \textsc{Swendsen}--\textsc{Wang}
        \item Mais un \emph{seul} tirage d'horloges contient désormais \textbf{toutes les échelles} intermédiaires :
        les $\Pi_\beta$ sont emboîtées, elles forment un \textbf{dendrogramme} \citb{ii26}
        --- $\beta$ est un temps de filtration, pas une température
    \end{itemize}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Les deux bouts de l'arbre : \textsc{Glauber} et \textsc{Swendsen}--\textsc{Wang}}
    \begin{center}
        \resizebox{0.72\textwidth}{!}{\input{figs/dendro_horloges}}
    \end{center}

    \vspace{0.15cm}
    \small
    \begin{itemize}
        \setlength{\itemsep}{6pt}
        \item À chaque \textbf{nœud} de l'arbre, un \emph{bain thermique} exact : tirer $D \mid \sigma$,
        appliquer les bains à $D$ fixé, oublier $D$ --- tout programme de nœuds choisi \emph{indépendamment}
        de la configuration laisse la loi \emph{a posteriori} invariante (démontré en volume fini)
        \item \textbf{À une racine} (une composante de $\Pi_1$), retourner l'amas entier se fait à pile ou
        face : c'est \textsc{Swendsen}--\textsc{Wang}. \textbf{À une feuille}, la mise à jour porte sur un
        spin isolé : c'est un bain de \textsc{Glauber}
        \item Entre les deux, un bain à \textbf{quatre états} sur l'orientation \emph{relative} des deux fils :
        la famille \textbf{interpole} entre \textsc{Swendsen}--\textsc{Wang} et les dynamiques locales
    \end{itemize}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Quelle coupe choisir ? celle où les amas \textcolor{inria-rouge}{percolent}}
    \begin{center}
        \resizebox{0.54\textwidth}{!}{\input{figs/coupe_percolation}}
    \end{center}

    \vspace{0.05cm}
    \begin{center}
        \formulebox{\parbox{0.80\textwidth}{\centering\small
        densité d'arêtes ouvertes avant $\beta$ :
        $\;q_p(\beta) = p\left(1 - e^{-u_p \beta}\right)$, \; $u_p = \log\tfrac{p}{1-p}$ ;\\[3pt]
        \textbf{coupe critique} $\beta_c$ : $\;q_p(\beta_c) = q_c$, le seuil de percolation du réseau}}
    \end{center}

    \vspace{0.10cm}
    \small
    \begin{itemize}
        \setlength{\itemsep}{4pt}
        \item Seule une masse \textbf{macroscopique} transporte de l'information --- c'est ce que dit
        la borne de recouvrement de la partie précédente, évaluée à l'horizon $\beta = 1$
        \item Sur la grille triangulaire, au seuil de percolation de l'\emph{information}, le temps
        informationnel et $\beta_c$ \textbf{coïncident} \citb{ii26} : mon meilleur argument pour dire
        que c'est la bonne coupe
        \item \textbf{Statut : un programme.} Notre meilleure borne rigoureuse vient d'ailleurs
        --- d'un canal d'information à plusieurs états sur les triangles --- \emph{pas} de cette dynamique
    \end{itemize}
\end{frame}

\section{Conclusion}
\frame{\sectionpage}

\begin{frame}{Deux piliers, quatre contributions}
    \begin{beamercolorbox}[rounded=true,sep=8pt]{block body}
        \centering
        \textbf{1)} Quand les relations binaires sont insuffisantes ou instables,\\ les \textbf{interactions d'ordre supérieur} apportent une cohérence robuste.\\[6pt]
        \textbf{2)} La \textbf{percolation} est la clef de voûte théorique\\ pour quantifier les performances de ces algorithmes.
    \end{beamercolorbox}

    \vspace{0.20cm}
    \small
    \begin{itemize}
        \setlength{\itemsep}{5pt}
        \item Une \textbf{vue unifiée} du \emph{Single-Linkage} : heuristique (arbre couvrant), géométrique (percolation), statistique (\textsc{Hartigan}) ; applications LiDAR 3D et 4D sans apprentissage
        \item \textbf{HGP-Clusterer} : la hiérarchie $K$-NN \emph{exacte}, \emph{via} \v{C}ech, \textsc{Gabriel} et la mosaïque de \textsc{Delaunay} d'ordre $K$ \citb{eusipco24} \citb{cn24} \citb{ans26}
        \item La \textbf{vitesse de percolation} : une mesure quantitative qui explique l'échec de Robust SL et (H)DBSCAN et les gains observés des $K$-polyèdres dans ce protocole \citb{src24}
        \item L'\textbf{export du paradigme} : graphes signés (dynamiques d'ordre supérieur et bornes de recouvrement) \citb{asilomar25} \citb{ii26} ; imagerie (graphe de \textsc{Frangi} multimodal) \citb{gretsi25} \citb{euvip26}
    \end{itemize}
\end{frame}

\begin{frame}{Perspectives, dans le fil de la thèse}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Réduction de dimension d'ordre supérieur} : la mosaïque de \textsc{Delaunay} d'ordre $K$ est bien plus riche que les graphes de voisinage (UMAP) : la piste privilégiée à court terme
        \item \textbf{Question ouverte assumée} : HGP-Clusterer n'est que \emph{fractionnellement} consistant ; un algorithme consistant en dimension $p \ge 2$ reste à inventer
        \item \textbf{Graphes signés} : couplage à deux répliques pour des seuils exacts de recouvrement ; vers un algorithme pratique de détection de communautés
        \item \textbf{Fissures} : un logiciel d'aide à l'annotation experte fondé sur les graphes ; réconcilier modèles géométriques et apprentissage profond
        \item \textbf{Et, plus loin} : la \emph{voie hiérarchique} qui précède --- faire du polyèdre et de son espace d'échelle l'unité de calcul d'un modèle appris pour la 3D
    \end{itemize}
\end{frame}

\begin{frame}{Le cadre bayésien et l'hypothèse de Gibbs}
    \begin{center}
        \begin{tikzpicture}[node distance=0.7cm and 0.9cm, scale=0.85, every node/.style={scale=0.85}]
            \node[chapbox] (prior) {\emph{A priori}\\$\mu_{0}$};
            \node[chapbox,right=of prior] (truth) {Communautés\\cachées $\Sigma$};
            \node[chapbox,right=of truth] (chan) {Canal bruité\\$Q$};
            \node[chapboxdark,right=of chan] (obs) {Observation\\$(X,W)$};
            \node[chapbox,right=of obs] (post) {\emph{A posteriori}\\$\mu$};
            \draw[chaparr] (prior) -- (truth);
            \draw[chaparr] (truth) -- (chan);
            \draw[chaparr] (chan) -- (obs);
            \draw[chaparr] (obs) -- (post);
        \end{tikzpicture}
    \end{center}

    \vspace{0.3cm}
    \begin{center}
        \formulebox{$U(\sigma) = \displaystyle\sum_{e \in F} W_e \ind{\sigma_i \neq \sigma_j} + \sum_{e \in \tilde F} |W_e| \ind{\sigma_i = \sigma_j}
        \qquad
        \mu(\sigma) = \frac{e^{-U(\sigma)}}{Z}$}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \item $F$ : arêtes attractives ($W_e > 0$) ; $\tilde F$ : répulsives ($W_e < 0$) ; $U$ pénalise les arêtes non satisfaites
        \item Ici l'\emph{a priori} est uniforme : le terme $-\log\mu_0(\sigma \mid X)$ est constant et absorbé dans $Z$
        \item \textbf{Hypothèse de Gibbs} : la loi \emph{a posteriori} est la mesure de Gibbs signée
    \end{itemize}
\end{frame}

\begin{frame}{Un pas de Swendsen--Wang signé, en détail}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Gel} : chaque arête \emph{satisfaite} est gelée avec probabilité $1 - e^{-|W_e|}$ ; les non satisfaites jamais
        \item \textbf{Recoloriage} : chaque amas gelé est recolorié uniformément au hasard
        \item La dynamique laisse la loi \emph{a posteriori} invariante
        \item Un pas depuis la vérité $\Sigma$ produit une \textbf{réplique} $\Sigma^{(1)} \sim \mu$ (identité de Nishimori) : un couplage entre vérité et observation
        \item Les petits amas sont recoloriés aveuglément : l'information globale ne survit que dans les \textbf{composantes géantes}
    \end{itemize}
\end{frame}

\begin{frame}{Le modèle de Sankararaman--Baccelli, en détail}
    \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Nœuds : processus de Poisson dans $\mathbb R^p$ ; communautés $\Sigma_i \in \{\pm 1\}$ uniformes
        \item Arête $e = \{i,j\}$ présente avec probabilité $f_{\mathrm{in}}(r_e)$ (même communauté) ou $f_{\mathrm{out}}(r_e)$ (communautés différentes), $r_e = \|X_i - X_j\|$
        \item Poids signés reconstruits : $W_e = \log\frac{f_{\mathrm{in}}(r_e)}{f_{\mathrm{out}}(r_e)} \ge 0$ si $e$ observée, $W_e = -\log\frac{1-f_{\mathrm{out}}(r_e)}{1-f_{\mathrm{in}}(r_e)} \le 0$ sinon
    \end{itemize}

    \vspace{0.3cm}
    \begin{center}
        \formulebox{$\mathbb P(\text{arête gelée}) = f_{\mathrm{in}}(r_e) \left(1 - e^{-|W_e|}\right) = f_{\mathrm{in}}(r_e) - f_{\mathrm{out}}(r_e)$}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \item Le graphe gelé d'un pas de SW sur les arêtes est \textbf{exactement} la percolation indépendante de \citb{sb18}
    \end{itemize}
\end{frame}

\begin{frame}{La hiérarchie d'horloges, en détail}
    \begin{itemize}
        \setlength{\itemsep}{8pt}
        \item Conditionnellement à $\sigma$ : $\xi_e \sim \mathrm{Exp}(|W_e|)$ sur les arêtes satisfaites, $\xi_e = +\infty$ sinon ;\\
        $\Pi_\beta$ = composantes de $\{\xi_e \le \beta\}$ ; le dendrogramme s'obtient par \textsc{Kruskal}
        \item $\mathbb P(\xi_e \le 1) = 1 - e^{-|W_e|}$ : $\Pi_1$ est \textbf{exactement} la partition gelée de Swendsen--Wang
        \item Mesure jointe de type \textsc{Edwards}--\textsc{Sokal} $\nu(\sigma, D)$ ; à chaque nœud, un \textbf{bain thermique exact} sur les quatre orientations relatives des deux fils (balance détaillée, invariance)
        \item Le dendrogramme est \textbf{non marqué} : l'identité de l'arête gagnante est marginalisée --- une fusion dépend de toute la coupe $\Lambda_u(\sigma) = \sum_{e \in E_u} |W_e| \ind{e \text{ satisfaite}}$
        \item Sur le tore triangulaire : $\Pi_t$ suit une percolation indépendante $q_p(t) = p\,(1 - e^{-u_p t})$, $u_p = \log\frac{p}{1-p}$
    \end{itemize}
\end{frame}

\begin{frame}{La coupe critique $\beta_c$, en détail}
    \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Le niveau critique s'inverse explicitement :
        \[
            \beta_c(p) \;=\; q_p^{-1}(q_c) \;=\; -\,\frac{\log\bigl(1 - q_c/p\bigr)}{\log\frac{p}{1-p}},
            \qquad q_c = 2\sin\tfrac{\pi}{18} = 0{,}34730\ldots
        \]
        \item Il n'existe dans l'horizon $[0,1]$ que si $p \ge \tfrac{1 + q_c}{2} = 0{,}67365\ldots$ ---
        c'est exactement le seuil de gel de \textsc{Swendsen}--\textsc{Wang} ; en dessous, la criticité n'est
        jamais atteinte avant la censure
        \item \textbf{Ce que la hiérarchie ajoute à la borne de percolation.} Celle-ci ne retient que la
        \emph{taille} des racines ; la borne par plus petit ancêtre commun pondère chaque fusion par sa
        fiabilité exacte --- algèbre finie établie, \emph{conditionnellement} à la formalisation complète
        de la mesure jointe augmentée
        \[
            Q_n \;\le\; \frac{1}{n^2}\,\mathbb E\Bigl[\,n + 2\!\!\sum_{u \in D}\! |C_{u,1}|\,|C_{u,2}|\,\eta_u\Bigr],
            \qquad \eta_u = \tanh^2\tfrac{L_u}{2}
        \]
        \item \textbf{Réserve.} La coupe critique n'est \emph{pas} sélectionnée par l'information --- après
        marginalisation exacte, le jacobien répliqué vaut $\theta^2$ à tout niveau. Elle se distingue
        \emph{géométriquement}, comme ordre d'élimination du \textsc{Gibbs}
    \end{itemize}
\end{frame}

\begin{frame}{Guider SAM : une suite progressive et réfutable}
    \begin{enumerate}
        \setlength{\itemsep}{12pt}
        \item \textbf{Mesurer le plafond, sur SAM~2 gelé.} Injecter la \emph{vérité terrain} dans l'interface
        choisie et regarder ce qu'elle rapporte au mieux. La vérité terrain à l'inférence en fait un
        \textbf{diagnostic}, jamais un résultat publiable --- mais deux heures de GPU suffisent, et
        sans plafond on arrête là.
        \item \textbf{Utiliser d'abord l'interface native.} Transformer la hiérarchie en invites ponctuelles,
        sélectionnées par centralité décroissante de l'arbre couvrant minimal, du grossier au fin.
        Ce serait la première utilisation réelle de l'arbre et des composantes.
        \item \textbf{Seulement si le signal est causal.} Apprendre un biais additif selon la distance dans
        le graphe ou dans l'arbre, initialisé à zéro : \textbf{informer} l'attention sans lier ses coefficients.
    \end{enumerate}

    \vfill
    \begin{center}
        \formulebox{\parbox{0.82\textwidth}{\centering\small
        À chaque étape : \textbf{aligné contre permuté et aléatoire},
        à capacité et budget identiques. Sans séparation, on arrête.}}
    \end{center}
\end{frame}
