\begin{frame}{En pratique : estimer la densité}
    \begin{center}

    \end{center}

    \vspace{0.4cm}

    \begin{columns}[c, onlytextwidth]
        \begin{column}{0.56\textwidth}
            \begin{center}
                {\large\mathversion{bold} $\forall x \in \mathbb{R}^p, \quad \widehat{f}_{K\text{-NN}}(x) = \dfrac{K}{n \, \omega_p \, r_K(x)^p}$}

                \vspace{1.0cm}

                \formulebox{\small\mathversion{bold} $r_K(x) = \min \left\{ r \ge 0 \;\middle|\; |B(x,r) \cap (\mathcal{X}\setminus\{x\})| \ge K \right\}$}

\begin{frame}{Le théorème d'impossibilité de \textsc{Kleinberg}}
    \small
    Soit $F$ une règle qui associe à toute dissimilarité $d$ sur un ensemble fini $X$, $|X|\ge2$,
    une \textbf{partition} $F(d)$ de $X$.

    \vspace{0.20cm}
    \begin{enumerate}
        \setlength{\itemsep}{9pt}
        \item \textbf{Richesse.} Toute partition $P$ de $X$ peut être obtenue :
        $\forall P,\ \exists d,\ F(d)=P$.
        \item \textbf{Invariance d'échelle.} Changer d'unité ne change pas la partition :
        $\forall \alpha>0,\ F(\alpha d)=F(d)$.
        \item \textbf{Cohérence.} Si $F(d)=P$, rapprocher les points d'un même bloc de $P$
        et éloigner ceux de blocs distincts doit laisser $P$ inchangée.
    \end{enumerate}

    \vfill
    \begin{center}
        \formulebox{\textbf{Impossibilité} \citb{kleinberg02} : aucune règle $F$ ne satisfait simultanément ces trois axiomes.}
    \end{center}
    \reffoot{kleinberg02}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Pourquoi la hiérarchie était la bonne solution}
    \begin{center}
        \formulebox{\parbox{0.82\textwidth}{\centering
        Une partition force une \textbf{échelle unique}.\\[3pt]
        Une hiérarchie rend \textbf{toutes les partitions emboîtées} :
        $\displaystyle T(d)=\{\theta_d(r)\}_{r\ge0}$,
        avec $r\le r'\Rightarrow\theta_d(r)\preceq\theta_d(r')$.}}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \setlength{\itemsep}{7pt}
        \item On ne demande plus à l'algorithme de choisir arbitrairement une coupe :
        la résolution $r$ reste visible dans la sortie.
        \item Un changement d'unité \textbf{reparamètre} l'arbre au lieu de l'effacer :
        $\theta_{\alpha d}(r)=\theta_d(r/\alpha)$ ; sa structure emboîtée est préservée.
        \item Le théorème n'est pas contredit : on \textbf{change l'objet cible}.
        Pour les méthodes hiérarchiques à sortie ultramétrique, des axiomes adaptés caractérisent alors le
        \textbf{\emph{Single-Linkage}} \citb{carlsson10}.
    \end{itemize}
    \reffoot{kleinberg02,carlsson10}
\end{frame}

\begin{frame}{Une suite simple, progressive et réfutable}
    \begin{enumerate}
        \setlength{\itemsep}{12pt}
        \item \textbf{Test idéalisé sur SAM~2 gelé.} Injecter d'abord la vérité terrain dans
        l'interface choisie pour vérifier si la hiérarchie offre seulement une marge.
        La vérité terrain à l'inférence en fait un \textbf{diagnostic}, jamais un résultat publiable.
        \item \textbf{Utiliser d'abord l'interface native.} Transformer la hiérarchie en
        invites ponctuelles, sélectionnées par centralité de l'arbre couvrant minimal,
        du grossier au fin.
        \item \textbf{Seulement si le signal est causal.} Apprendre un biais additif résiduel
        selon la distance dans le graphe ou l'arbre, initialisé à zéro :
        \textbf{informer} l'attention sans lier ses coefficients.
    \end{enumerate}

    \vfill
    \begin{center}
        \formulebox{\parbox{0.82\textwidth}{\centering\small
        À chaque étape : \textbf{aligné contre permuté et aléatoire},
        à capacité et budget identiques. Sans séparation, on arrête.}}
    \end{center}

\end{frame}

\section{Conclusion}
\frame{\sectionpage}

\begin{frame}{Deux piliers, quatre contributions}
    \begin{beamercolorbox}[rounded=true,sep=8pt]{block body}
        \centering
        \textbf{1)} Quand les relations binaires sont insuffisantes ou instables,\\ les \textbf{interactions d'ordre supérieur} apportent une cohérence robuste.\\[6pt]
        \textbf{2)} La \textbf{percolation} est la clef de voûte théorique\\ pour quantifier les performances de ces algorithmes.
    \end{beamercolorbox}

    \vspace{0.20cm}
    \small
    \begin{itemize}
        \setlength{\itemsep}{5pt}
        \item Une \textbf{vue unifiée} du \emph{Single-Linkage} : heuristique (arbre couvrant), géométrique (percolation), statistique (\textsc{Hartigan}) ; applications LiDAR 3D et 4D sans apprentissage
        \item \textbf{HGP-Clusterer} : la hiérarchie $K$-NN \emph{exacte}, \emph{via} \v{C}ech, \textsc{Gabriel} et la mosaïque de \textsc{Delaunay} d'ordre $K$ \citb{eusipco24} \citb{cn24} \citb{ans26}
        \item La \textbf{vitesse de percolation} : une mesure quantitative qui explique l'échec de Robust SL et (H)DBSCAN et les gains observés des $K$-polyèdres dans ce protocole \citb{src24}
        \item L'\textbf{export du paradigme} : graphes signés (dynamiques d'ordre supérieur et bornes de recouvrement) \citb{asilomar25} \citb{ii26} ; imagerie (graphe de \textsc{Frangi} multimodal) \citb{gretsi25} \citb{euvip26}
    \end{itemize}

\begin{frame}{Le cadre bayésien et l'hypothèse de Gibbs}
    \begin{center}
        \begin{tikzpicture}[node distance=0.7cm and 0.9cm, scale=0.85, every node/.style={scale=0.85}]
            \node[chapbox] (prior) {\emph{A priori}\\$\mu_{0}$};
            \node[chapbox,right=of prior] (truth) {Vérité cachée\\$\Sigma$};
            \node[chapbox,right=of truth] (chan) {Canal bruité\\$Q$};
            \node[chapboxdark,right=of chan] (obs) {Observation\\$(X,W)$};
            \node[chapbox,right=of obs] (post) {\emph{A posteriori}\\$\mu_{X,W}$};
            \draw[chaparr] (prior) -- (truth);
            \draw[chaparr] (truth) -- (chan);
            \draw[chaparr] (chan) -- (obs);
            \draw[chaparr] (obs) -- (post);
        \end{tikzpicture}
    \end{center}

    \vspace{0.3cm}
    \begin{center}
        \formulebox{$U(\sigma) = \displaystyle\sum_{e \in F} W_e \ind{\sigma_i \neq \sigma_j} + \sum_{e \in \tilde F} |W_e| \ind{\sigma_i = \sigma_j}
        \qquad
        \mu_{X,W}(\sigma) = \frac{e^{-U(\sigma)}}{Z}$}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \item $F$ : arêtes attractives ($W_e > 0$) ; $\tilde F$ : répulsives ($W_e < 0$) ; $U$ pénalise les arêtes insatisfaites
        \item \textbf{Hypothèse de Gibbs} : la loi \emph{a posteriori} est la mesure de Gibbs signée
    \end{itemize}
\end{frame}

\begin{frame}{Un pas de Swendsen--Wang signé, en détail}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{Gel} : chaque arête \emph{satisfaite} est gelée avec probabilité $1 - e^{-|W_e|}$ ; les insatisfaites jamais
        \item \textbf{Recoloriage} : chaque amas gelé est recolorié uniformément au hasard
        \item La dynamique laisse la loi \emph{a posteriori} invariante
        \item Un pas depuis la vérité $\Sigma$ produit une \textbf{réplique} $\Sigma^{(1)} \sim \mu_{X,W}$ (identité de Nishimori) : un couplage entre vérité et observation
        \item Les petits amas sont recoloriés aveuglément : l'information globale ne survit que dans les \textbf{composantes géantes}
    \end{itemize}
\end{frame}

\begin{frame}{Le modèle de Sankararaman--Baccelli, en détail}
    \begin{itemize}
        \setlength{\itemsep}{9pt}
        \item Nœuds : processus de Poisson dans $\mathbb R^p$ ; communautés $\Sigma_i \in \{\pm 1\}$ uniformes
        \item Arête $e = \{i,j\}$ présente avec probabilité $f_{\mathrm{in}}(r_e)$ (même communauté) ou $f_{\mathrm{out}}(r_e)$ (communautés différentes), $r_e = \|X_i - X_j\|$
        \item Poids signés reconstruits : $W_e = \log\frac{f_{\mathrm{in}}(r_e)}{f_{\mathrm{out}}(r_e)} \ge 0$ si $e$ observée, $W_e = -\log\frac{1-f_{\mathrm{out}}(r_e)}{1-f_{\mathrm{in}}(r_e)} \le 0$ sinon
    \end{itemize}

    \vspace{0.3cm}
    \begin{center}
        \formulebox{$\mathbb P(\text{arête gelée}) = f_{\mathrm{in}}(r_e) \left(1 - e^{-|W_e|}\right) = f_{\mathrm{in}}(r_e) - f_{\mathrm{out}}(r_e)$}
    \end{center}

    \vspace{0.25cm}
    \begin{itemize}
        \item Le graphe gelé d'un pas de SW sur les arêtes est \textbf{exactement} la percolation indépendante de \citb{sb18}
    \end{itemize}
\end{frame}

\begin{frame}{La hiérarchie d'horloges, en détail}
    \begin{itemize}
        \setlength{\itemsep}{8pt}
        \item Conditionnellement à $\sigma$ : $\xi_e \sim \mathrm{Exp}(|W_e|)$ sur les arêtes satisfaites, $\xi_e = +\infty$ sinon ;\\
        $\Pi_\beta$ = composantes de $\{\xi_e \le \beta\}$ ; le dendrogramme s'obtient par \textsc{Kruskal}
        \item $\mathbb P(\xi_e \le 1) = 1 - e^{-|W_e|}$ : $\Pi_1$ est \textbf{exactement} la partition gelée de Swendsen--Wang
        \item Mesure jointe de type \textsc{Edwards}--\textsc{Sokal} $\nu(\sigma, D)$ ; à chaque nœud, un \textbf{bain thermique exact} sur les quatre orientations relatives des deux fils (balance détaillée, invariance)
        \item Le dendrogramme est \textbf{non marqué} : l'identité de l'arête gagnante est marginalisée --- une fusion dépend de toute la coupe $\Lambda_u(\sigma) = \sum_{e \in E_u} |W_e| \ind{e \text{ satisfaite}}$
        \item Sur le tore triangulaire : $\Pi_t$ suit une percolation indépendante $q_p(t) = p\,(1 - e^{-u_p t})$, $u_p = \log\frac{p}{1-p}$
    \end{itemize}
\end{frame}

        \item \textbf{Surface d'attaque} : $3$ blocs d'attention globale sur $48$ dans Hiera-L, soit $6{,}56\,\%$ des opérations du tronc
        \item \textbf{Forme de l'arbre} : ce n'est pas un réglage --- l'arité $b = (N-1)/(N-L)$ est \emph{fixée} par la fraction de feuilles, donc par la géométrie de la fissure
        \item Ce défaut-là se répare (décomposition en centroïdes de l'arbre couvrant minimal \emph{non élagué}, sur la grille $64 \times 64$ : profondeur $12$, arité $2{,}82$) ; ce qui ne se répare pas, c'est qu'une contrainte de blocs \emph{comprime} au lieu d'informer
        \item \textbf{Limite assumée} : mesures sur fissures \emph{synthétiques} calibrées sur le corpus Khánh Hà ($448$~px, largeur $19$~px, couverture $5{,}7\,\%$)
    \end{itemize}
\end{frame}
